    @router.get("/me", summary="Validate credential; return identity and topics")
async def me(request: Request, api_key: ApiKey) -> MeResponse:
    state: FirehoseState = request.app.state.firehose
    identity = await state.auth_repo.identify(api_key=api_key)
    if not identity.allowed:
        logger.info(f"credential rejected user={identity.user}")
        return MeResponse(user=identity.user, token=None, allowed=False, topics=[])
    if identity.user == ANONYMOUS_USER:
        topics = [t for t in ALLOWED_TOPICS if t not in state.keyed_only_topics]
    else:
        topics = firehose_topics(identity.scopes)
    return MeResponse(user=identity.user, token=api_key, allowed=True, topics=topics)

    if WILDCARD_SCOPE in scopes:
        return ALLOWED_TOPICS
    return [
        scope[len(SCOPE_PREFIX) :]

close codes

The route accepts the handshake before the gates, so every rejection below arrives as a websocket close frame carrying these codes — not as an http error.

codereasonwhen
4404unknown topictopic is not in the TOPIC_MODELS catalog
4401invalid api keytoken rejected, identity not allowlisted, or the firehose.<topic> scope is missing
4403topic requires an api keyanonymous connect to a keyed-only topic
1011internal errorconsumer creation failed
1013slow consumerbackpressure kick after the join grace (see the base README)
